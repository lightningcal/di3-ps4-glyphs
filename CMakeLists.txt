cmake_minimum_required(VERSION 3.21)
project(DI3PS4Glyphs VERSION 0.1.5 LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
    message(FATAL_ERROR "Build for Windows Win32 (x86), not x64")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(PS4Glyphs SHARED src/plugin.cpp)
set_target_properties(PS4Glyphs PROPERTIES PREFIX "" SUFFIX ".asi")
target_include_directories(PS4Glyphs PRIVATE third_party src)
target_compile_definitions(PS4Glyphs PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
target_compile_options(PS4Glyphs PRIVATE /W4 /EHsc)
target_link_libraries(PS4Glyphs PRIVATE d3d11 windowscodecs ole32)
enable_testing()
add_executable(test_fingerprint tests/test_fingerprint.cpp)
target_include_directories(test_fingerprint PRIVATE src)
add_test(NAME fingerprint COMMAND test_fingerprint)

find_program(CRABE_LUA NAMES crabe_lua lua51 lua5.1 lua
    HINTS "${CMAKE_CURRENT_SOURCE_DIR}/../CrabeLoader/build/vs2022-dll/Release")
if(CRABE_LUA)
    add_test(NAME lua_settings COMMAND "${CRABE_LUA}" tests/test_settings.lua)
    set_tests_properties(lua_settings PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
endif()
add_executable(test_renderer tests/test_renderer.cpp)
target_include_directories(test_renderer PRIVATE third_party)
target_compile_definitions(test_renderer PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
target_link_libraries(test_renderer PRIVATE d3d11 d3dcompiler)

add_custom_command(TARGET PS4Glyphs POST_BUILD
    COMMAND "${CMAKE_COMMAND}" -E copy_directory
    "${CMAKE_CURRENT_SOURCE_DIR}/mod/glyphs" "$<TARGET_FILE_DIR:PS4Glyphs>/glyphs")
set(PS4_GLYPH_TEST_DDS "" CACHE FILEPATH "Optional local vanilla glyph DDS for the real GPU integration test")
if(EXISTS "${PS4_GLYPH_TEST_DDS}")
    add_test(NAME renderer COMMAND test_renderer "$<TARGET_FILE:PS4Glyphs>" "${PS4_GLYPH_TEST_DDS}")
endif()
